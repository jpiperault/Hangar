// 1. Au démarrage (Application.onCreate) : configuration à distance, en arrière-plan
val CONFIG = "{{ rc_urls.android or 'https://apps./.well-known/apps/' ~ a.pkg ~ '.json' }}"
thread {
    runCatching {
        val c = JSONObject(URL(CONFIG).readText())
        prefs.edit().putString("hangar_config", c.toString()).apply()   // relue au démarrage suivant si pas de réseau
        val code = BuildConfig.VERSION_CODE
        if (!c.isNull("min_version_code") && code < c.getInt("min_version_code")) { /* écran « Mets à jour » bloquant → c.getString("store_url") */ }
        else if (!c.isNull("recommended_version_code") && code < c.getInt("recommended_version_code")) { /* proposer la mise à jour */ }
        val flags = c.getJSONObject("flags")          // ex. flags.optBoolean("nouveau_menu")
        if (!c.isNull("message")) { /* afficher c.getJSONObject("message").getString("text") */ }
    }
}

// 2. Plantages : écrits au moment du plantage, envoyés au démarrage suivant
val prev = Thread.getDefaultUncaughtExceptionHandler()
Thread.setDefaultUncaughtExceptionHandler { t, e ->
    runCatching {
        File(filesDir, "crash.json").writeText(JSONObject()
            .put("version_name", BuildConfig.VERSION_NAME).put("version_code", BuildConfig.VERSION_CODE)
            .put("os", "Android ${Build.VERSION.RELEASE}").put("device", "${Build.MANUFACTURER} ${Build.MODEL}")
            .put("exception", e.javaClass.name).put("message", e.message ?: "")
            .put("stacktrace", e.stackTraceToString().take(15000)).toString())
    }
    prev?.uncaughtException(t, e)
}
thread {
    val f = File(filesDir, "crash.json")
    val c = runCatching { JSONObject(prefs.getString("hangar_config", "{}")!!) }.getOrNull()
    if (f.exists() && c != null && c.has("crash_url")) runCatching {
        (URL(c.getString("crash_url")).openConnection() as HttpURLConnection).run {
            requestMethod = "POST"; doOutput = true; connectTimeout = 5000; readTimeout = 5000
            setRequestProperty("Content-Type", "application/json")
            setRequestProperty("X-Hangar-Key", c.getString("crash_key"))
            outputStream.use { it.write(f.readBytes()) }
            if (responseCode in 200..299 || responseCode == 400 || responseCode == 413) f.delete()
        }
    }
}

// 1. Au lancement : configuration à distance (en arrière-plan, mémorisée dans UserDefaults)
let configURL = URL(string: "{{ rc_urls.ios or 'https://apps./.well-known/apps/' ~ a.ios_bundle ~ '.json' }}")!
Task.detached {
    if let (data, _) = try? await URLSession.shared.data(from: configURL),
       let c = try? JSONSerialization.jsonObject(with: data) as? [String: Any] {
        UserDefaults.standard.set(data, forKey: "hangar_config")
        let build = Int(Bundle.main.infoDictionary?["CFBundleVersion"] as? String ?? "0") ?? 0
        if let min = c["min_version_code"] as? Int, build < min { /* écran « Mets à jour » → c["store_url"] */ }
        else if let rec = c["recommended_version_code"] as? Int, build < rec { /* proposer la mise à jour */ }
        let flags = c["flags"] as? [String: Any] ?? [:]   // ex. flags["nouveau_menu"] as? Bool
    }
}

// 2. Plantages (exceptions Objective-C) : écrits dans un fichier, envoyés au lancement suivant
let crashFile = FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0].appendingPathComponent("crash.json")
NSSetUncaughtExceptionHandler { e in
    let info = Bundle.main.infoDictionary ?? [:]
    let r: [String: Any] = ["exception": e.name.rawValue, "message": e.reason ?? "",
        "stacktrace": e.callStackSymbols.joined(separator: "\n"),
        "version_name": info["CFBundleShortVersionString"] ?? "", "version_code": Int(info["CFBundleVersion"] as? String ?? "0") ?? 0,
        "os": "iOS \(UIDevice.current.systemVersion)", "device": UIDevice.current.model]
    try? JSONSerialization.data(withJSONObject: r).write(to: crashFile)
}
if let body = try? Data(contentsOf: crashFile),
   let saved = UserDefaults.standard.data(forKey: "hangar_config"),
   let c = try? JSONSerialization.jsonObject(with: saved) as? [String: Any],
   let url = (c["crash_url"] as? String).flatMap(URL.init), let key = c["crash_key"] as? String {
    var req = URLRequest(url: url); req.httpMethod = "POST"; req.httpBody = body
    req.setValue("application/json", forHTTPHeaderField: "Content-Type"); req.setValue(key, forHTTPHeaderField: "X-Hangar-Key")
    URLSession.shared.dataTask(with: req) { _, resp, _ in
        if let code = (resp as? HTTPURLResponse)?.statusCode, (200..<300).contains(code) || code == 400 || code == 413 {
            try? FileManager.default.removeItem(at: crashFile) }
    }.resume()
}
